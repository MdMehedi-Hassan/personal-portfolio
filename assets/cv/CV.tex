\section{Research Projects}

\entry{RailTrack Inspector -- Railway Defect Detection System}{\href{https://doi.org/10.1109/QPAIN69676.2026.11546344}{IEEE DOI}}
\begin{itemize}
    \item Developed an explainable deep learning-based railway inspection system to detect missing fasteners, missing bolts, and rail cracks, addressing the limitations of traditional manual inspection.
    \item Integrated Grad-CAM for visual interpretation of model predictions and GPS-enabled data collection for geolocation-based defect mapping.
    \item Implemented Faster R-CNN, achieving 97\% accuracy, and deployed the trained model through a Streamlit web application for automated defect detection and visualization.
    \item \textit{Technologies:} Python, Faster R-CNN, Grad-CAM, Streamlit, GPS-based Localization, Deep Learning.
    \item \textit{Links:} \href{https://doi.org/10.1109/QPAIN69676.2026.11546344}{Publication} \;|\;
    \href{https://github.com/MdMehedi-Hassan/RailTrack-Inspector}{GitHub}
\end{itemize}
